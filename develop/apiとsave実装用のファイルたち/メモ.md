# スコア送信（POST）の非同期本体
async def _post_score_async(player_name, score):
    try:
        await pyfetch(
            GAS_URL,
            method="POST",
            headers={"Content-Type": "text/plain"},
            body=json.dumps({"name": player_name, "score": score}),
        )
    except Exception as e:
        print("スコア送信エラー:", e)

# 同期関数から呼び出すためのラッパー（既存の呼び出し方のまま使えます）
